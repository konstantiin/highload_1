@startuml
!include https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Component.puml

LAYOUT_WITH_LEGEND()

skinparam nodesep 90
skinparam ranksep 90

Person(trader, "Трейдер", "Регистрируется, проходит KYC, получает токен доступа и размещает ордера")
Person(admin, "Администратор платформы", "Аутентифицируется, получает токен доступа, проверяет и одобряет/отклоняет данные KYC")
Person(regulator, "Регулятор", "Аутентифицируется, получает токен доступа и проверяет пользователей, данные KYC, балансы и журналы")

Container_Boundary(platform, "Торговая платформа") {

    Component(auth, "Сервис авторизации", "Идентификация / учетные записи", "Хранит учетные данные пользователей, балансы и дополнительные данные KYC, такие как фотографии, документы и текст. Управляет одобрением KYC администраторами и выдает токены доступа.")

    Component(trading, "Торговый компонент", "Управление ордерами / сопоставление ордеров", "Принимает и отслеживает размещенные ордера. Исполняет сделки, если ордера исполняемы, и инициирует обновление балансов пользователей, разместивших ордера.")

    Component(market, "Сервис рынка", "Симулируемый маркет-мейкер", "Создает искусственные ордера, чтобы цена финансового инструмента колебалась. Объем и цена искусственных ордеров определяются объемом реальных ордеров и средней стоимостью сделки.")

    Component(logging, "Сервис журналирования", "Журнал событий / аудита", "Хранит события, предупреждения, попытки входа, торговые журналы и другие аудиторские данные. Поддерживает эффективные запросы по времени и тегам.")
}


Rel_U(trading, auth, "Проверяет токен доступа и читает/обновляет баланс пользователя")

Rel_U(market, trading, "Читает объем реальных ордеров, среднюю стоимость сделки и подает искусственные ордера")

Rel_D(auth, logging, "Пишет события входа, безопасности и одобрения/отклонения KYC")
Rel_D(trading, logging, "Пишет события ордеров, исполнения сделок и изменения балансов")
Rel_D(market, logging, "Пишет события искусственных ордеров и предупреждения")

Rel_R(trader, auth, "Регистрируется, входит и получает токен доступа", "HTTPS")
Rel_R(trader, trading, "Размещает, отменяет и запрашивает ордера с токеном доступа", "HTTPS")

Rel_R(admin, auth, "Аутентифицируется, получает токен и управляет одобрением/отклонением KYC", "HTTPS")
Rel_D(admin, logging, "Запрашивает журналы по времени и тегам с токеном доступа", "HTTPS")

Rel_L(regulator, auth, "Аутентифицируется, получает токен и запрашивает данные пользователей, KYC, балансы и одобрения", "HTTPS")
Rel_D(regulator, logging, "Запрашивает аудиторские журналы, журналы входа, события KYC и торговли по времени и тегам", "HTTPS")
@enduml
